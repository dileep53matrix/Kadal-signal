24:     fishing_depth_m REAL,             -- from CMFRI: usual fishing depth
25:     facilities TEXT DEFAULT '{}',     -- JSON: {"ice":true,"fuel":false,"shelter":true,"auction":true}

28:     last_verified TEXT                -- ISO timestamp of last matching report
29: );
30: 
31: CREATE TABLE IF NOT EXISTS contributions (
32:     id INTEGER PRIMARY KEY AUTOINCREMENT,
33:     site_id TEXT NOT NULL REFERENCES sites(id),
34:     reporter_id TEXT NOT NULL,        -- demo: made-up names only, no personal data
35:     lat REAL, lon REAL,
36:     facilities TEXT NOT NULL,         -- JSON, same keys as sites.facilities
37:     distance_m REAL,                  -- distance from the reporter to the site (if GPS given)

39:     photo_path TEXT,
40:     created_at TEXT NOT NULL
41: );
42: 
43: CREATE TABLE IF NOT EXISTS boats (
44:     id INTEGER PRIMARY KEY AUTOINCREMENT,

46:     site_id TEXT NOT NULL REFERENCES sites(id),
47:     status TEXT DEFAULT 'in_port',    -- in_port / at_sea
48:     last_departed TEXT,
49:     last_returned TEXT
50: );
51: 
52: CREATE TABLE IF NOT EXISTS alert_state (
53:     id INTEGER PRIMARY KEY CHECK (id = 1),   -- one row: the INCOIS high-wave alert for Karnataka
54:     state TEXT NOT NULL,                     -- active / none  (anything else = unknown)
55:     set_at TEXT NOT NULL,
56:     valid_until TEXT NOT NULL,
57:     note TEXT,
58:     source TEXT                              -- manual (read from the INCOIS page by a person)
59: );

65: );
66: 
67: CREATE TABLE IF NOT EXISTS signals (
68:     id INTEGER PRIMARY KEY AUTOINCREMENT,
69:     site_id TEXT NOT NULL REFERENCES sites(id),

71:     reason_en TEXT NOT NULL,

79: def connect():
80:     DB_PATH.parent.mkdir(exist_ok=True)
81:     con = sqlite3.connect(DB_PATH)
82:     con.row_factory = sqlite3.Row
83:     return con
84: 
85: NEW_SITE_COLUMNS = {"cmfri_code": "TEXT", "taluk": "TEXT", "gears": "TEXT",
86:                     "operation_period": "TEXT", "colocated_with": "TEXT"}
87: 
88: NEW_CONTRIB_COLUMNS = {"distance_m": "REAL", "counted": "INTEGER DEFAULT 1"}
89: 
90: def _migrate(con, table, wanted):
91:     have = {r["name"] for r in con.execute(f"PRAGMA table_info({table})")}
92:     for col, typ in wanted.items():
93:         if col not in have:
94:             con.execute(f"ALTER TABLE {table} ADD COLUMN {col} {typ}")
95: 
96: def init():
97:     con = connect()
98:     con.executescript(SCHEMA)
99:     _migrate(con, "sites", NEW_SITE_COLUMNS)              # databases made in Step 1 / 2
100:     _migrate(con, "contributions", NEW_CONTRIB_COLUMNS)
101:     con.commit()
102:     return con
